from pydantic import BaseModel
from typing import List, Optional, Dict, Any
from enum import Enum

class ConversationState(str, Enum):
    """對話狀態"""
    INITIAL = "initial"
    CLARIFYING = "clarifying"
    CONFIRMING = "confirming"
    ANSWERING = "answering"
    COMPLETED = "completed"

class Message(BaseModel):
    """訊息模型"""
    role: str  # user, assistant, system
    content: str
    metadata: Optional[Dict[str, Any]] = {}

class ConversationRequest(BaseModel):
    """對話請求"""
    session_id: str
    message: str

class ConversationResponse(BaseModel):
    """對話回應"""
    session_id: str
    message: str
    state: ConversationState
    suggestions: Optional[List[str]] = []
    is_complete: bool = False

class ConversationContext(BaseModel):
    """對話上下文"""
    session_id: str
    messages: List[Message] = []
    state: ConversationState = ConversationState.INITIAL
    clarification_count: int = 0
    user_intent: Optional[str] = None
    extracted_keywords: List[str] = []
    confidence_score: float = 0.0

import os
from typing import List, Dict
from langchain_openai import OpenAIEmbeddings
from langchain.text_splitter import RecursiveCharacterTextSplitter
from langchain_community.vectorstores import Chroma
from langchain.schema import Document
import chromadb
from dotenv import load_dotenv

load_dotenv()

class VectorStoreManager:
    def __init__(self, collection_name: str = "rag_documents"):
        self.embeddings = OpenAIEmbeddings()
        self.collection_name = collection_name
        
        # 初始化 ChromaDB
        self.client = chromadb.PersistentClient(path="./chroma_db")
        
        # 初始化向量存儲
        self.vector_store = Chroma(
            client=self.client,
            collection_name=self.collection_name,
            embedding_function=self.embeddings
        )
        
        # 文本分割器
        self.text_splitter = RecursiveCharacterTextSplitter(
            chunk_size=500,
            chunk_overlap=50
        )
    
    def add_documents(self, texts: List[str], metadatas: List[Dict] = None):
        """添加文檔到向量存儲"""
        documents = []
        for i, text in enumerate(texts):
            metadata = metadatas[i] if metadatas else {}
            documents.append(Document(page_content=text, metadata=metadata))
        
        # 分割文檔
        split_docs = self.text_splitter.split_documents(documents)
        
        # 添加到向量存儲
        self.vector_store.add_documents(split_docs)
        
    def search(self, query: str, k: int = 5) -> List[Document]:
        """搜尋相關文檔"""
        return self.vector_store.similarity_search(query, k=k)
    
    def search_with_score(self, query: str, k: int = 5) -> List[tuple]:
        """搜尋並返回分數"""
        return self.vector_store.similarity_search_with_score(query, k=k)

6. conversation_graph.py
from typing import Dict, Any, List
from langgraph.graph import Graph, END
from langchain_openai import ChatOpenAI
from langchain.schema import HumanMessage, AIMessage, SystemMessage
import json
from models import ConversationState, ConversationContext, Message
from prompts import (
    INTENT_ANALYSIS_PROMPT, 
    CLARIFICATION_PROMPT,
    QUERY_REWRITE_PROMPT,
    ANSWER_PROMPT
)
from vector_store import VectorStoreManager

class ConversationGraph:
    def __init__(self, vector_store: VectorStoreManager):
        self.llm = ChatOpenAI(model="gpt-4", temperature=0.7)
        self.vector_store = vector_store
        self.graph = self._build_graph()
        
    def _build_graph(self) -> Graph:
        """構建對話流程圖"""
        workflow = Graph()
        
        # 添加節點
        workflow.add_node("analyze_intent", self.analyze_intent)
        workflow.add_node("generate_clarification", self.generate_clarification)
        workflow.add_node("rewrite_query", self.rewrite_query)
        workflow.add_node("retrieve_and_answer", self.retrieve_and_answer)
        
        # 添加邊
        workflow.add_edge("analyze_intent", self.route_after_analysis)
        workflow.add_edge("generate_clarification", END)
        workflow.add_edge("rewrite_query", "retrieve_and_answer")
        workflow.add_edge("retrieve_and_answer", END)
        
        # 設置入口點
        workflow.set_entry_point("analyze_intent")
        
        return workflow.compile()
    
    def analyze_intent(self, state: Dict[str, Any]) -> Dict[str, Any]:
        """分析使用者意圖"""
        context: ConversationContext = state["context"]
        
        # 準備訊息歷史
        messages = []
        for msg in context.messages:
            if msg.role == "user":
                messages.append(HumanMessage(content=msg.content))
            elif msg.role == "assistant":
                messages.append(AIMessage(content=msg.content))
        
        # 分析意圖
        prompt = INTENT_ANALYSIS_PROMPT.format_messages(
            messages=messages,
            input=state["current_input"]
        )
        
        response = self.llm.invoke(prompt)
        
        try:
            analysis = json.loads(response.content)
            
            state["intent_analysis"] = analysis
            state["context"].confidence_score = analysis.get("clarity_score", 0.0)
            state["context"].extracted_keywords = analysis.get("keywords", [])
            
            # 更新狀態
            if analysis.get("is_clear", False) and analysis.get("clarity_score", 0) > 0.7:
                state["context"].state = ConversationState.CONFIRMING
            else:
                state["context"].state = ConversationState.CLARIFYING
                
        except json.JSONDecodeError:
            state["intent_analysis"] = {
                "is_clear": False,
                "clarity_score": 0.3,
                "unclear_aspects": ["無法解析回應"],
                "keywords": [],
                "suggested_questions": []
            }
            state["context"].state = ConversationState.CLARIFYING
        
        return state
    
    def generate_clarification(self, state: Dict[str, Any]) -> Dict[str, Any]:
        """生成澄清問題"""
        context: ConversationContext = state["context"]
        analysis = state.get("intent_analysis", {})
        
        # 準備訊息歷史
        messages = []
        for msg in context.messages:
            if msg.role == "user":
                messages.append(HumanMessage(content=msg.content))
            elif msg.role == "assistant":
                messages.append(AIMessage(content=msg.content))
        
        # 生成澄清問題
        prompt = CLARIFICATION_PROMPT.format_messages(
            messages=messages,
            unclear_aspects=", ".join(analysis.get("unclear_aspects", [])),
            keywords=", ".join(analysis.get("keywords", []))
        )
        
        response = self.llm.invoke(prompt)
        
        state["clarification_question"] = response.content
        state["suggestions"] = analysis.get("suggested_questions", [])
        context.clarification_count += 1
        
        return state
    
    def rewrite_query(self, state: Dict[str, Any]) -> Dict[str, Any]:
        """改寫查詢以優化檢索"""
        context: ConversationContext = state["context"]
        
        # 準備訊息歷史
        messages = []
        for msg in context.messages:
            if msg.role == "user":
                messages.append(HumanMessage(content=msg.content))
            elif msg.role == "assistant":
                messages.append(AIMessage(content=msg.content))
        
        # 改寫查詢
        prompt = QUERY_REWRITE_PROMPT.format_messages(messages=messages)
        response = self.llm.invoke(prompt)
        
        state["rewritten_query"] = response.content
        return state
    
    def retrieve_and_answer(self, state: Dict[str, Any]) -> Dict[str, Any]:
        """檢索相關文檔並生成答案"""
        query = state.get("rewritten_query", state["current_input"])
        
        # 檢索相關文檔
        docs = self.vector_store.search(query, k=5)
        context_text = "\n\n".join([doc.page_content for doc in docs])
        
        # 準備訊息歷史
        messages = []
        for msg in state["context"].messages:
            if msg.role == "user":
                messages.append(HumanMessage(content=msg.content))
            elif msg.role == "assistant":
                messages.append(AIMessage(content=msg.content))
        
        # 生成答案
        prompt = ANSWER_PROMPT.format_messages(
            messages=messages,
            context=context_text,
            question=query
        )
        
        response = self.llm.invoke(prompt)
        
        state["final_answer"] = response.content
        state["context"].state = ConversationState.COMPLETED
        
        return state
    
    def route_after_analysis(self, state: Dict[str, Any]) -> str:
        """根據分析結果決定下一步"""
        if state["context"].state == ConversationState.CLARIFYING:
            return "generate_clarification"
        else:
            return "rewrite_query"
    
    def process_message(self, context: ConversationContext, user_input: str) -> Dict[str, Any]:
        """處理使用者訊息"""
        state = {
            "context": context,
            "current_input": user_input
        }
        
        result = self.graph.invoke(state)
        
        return result

@app.get("/")

@app.websocket("/ws/{session_id}")
async def websocket_endpoint(websocket: WebSocket, session_id: str):
    await websocket.accept()
    
    # 初始化或獲取對話上下文
    if session_id not in active_conversations:
        active_conversations[session_id] = ConversationContext(
            session_id=session_id,
            state=ConversationState.INITIAL
        )
    
    context = active_conversations[session_id]
    
    # 發送歡迎訊息
    await websocket.send_json({
        "message": "您好！我是您的智能助手。請告訴我您想了解什麼？",
        "state": context.state.value,
        "suggestions": ["如何使用 FastAPI？", "什麼是 RAG？", "LangChain 的功能"]
    })
    
    try:
        while True:
            # 接收使用者訊息
            data = await websocket.receive_json()
            user_message = data.get("message", "")
            
            if not user_message:
                continue
            
            # 添加使用者訊息到歷史
            context.messages.append(Message(role="user", content=user_message))
            
            # 處理訊息
            result = conversation_graph.process_message(context, user_message)
            
            # 根據結果更新上下文和生成回應
            updated_context = result["context"]
            
            if updated_context.state == ConversationState.CLARIFYING:
                # 發送澄清問題
                clarification = result.get("clarification_question", "能否請您更具體地說明您的問題？")
                suggestions = result.get("suggestions", [])
                
                context.messages.append(Message(role="assistant", content=clarification))
                
                await websocket.send_json({
                    "message": clarification,
                    "state": updated_context.state.value,
                    "suggestions": suggestions[:3],  # 最多顯示3個建議
                    "clarification_count": updated_context.clarification_count
                })
                
            elif updated_context.state == ConversationState.COMPLETED:
                # 發送最終答案
                answer = result.get("final_answer", "抱歉，我無法找到相關資訊。")
                
                context.messages.append(Message(role="assistant", content=answer))
                
                await websocket.send_json({
                    "message": answer,
                    "state": updated_context.state.value,
                    "is_complete": True,
                    "confidence_score": updated_context.confidence_score
                })
                
                # 重置對話狀態，準備下一輪
                context.state = ConversationState.INITIAL
                context.clarification_count = 0
            
            # 更新對話上下文
            active_conversations[session_id] = updated_context
            
    except WebSocketDisconnect:
        # 清理斷開連接的對話
        if session_id in active_conversations:
            del active_conversations[session_id]
        print(f"Session {session_id} disconnected")

@app.post("/api/conversation")
async def handle_conversation(request: ConversationRequest) -> ConversationResponse:
    """HTTP API 端點（可選）"""
    session_id = request.session_id
    
    if session_id not in active_conversations:
        active_conversations[session_id] = ConversationContext(
            session_id=session_id,
            state=ConversationState.INITIAL
        )
    
    context = active_conversations[session_id]
    context.messages.append(Message(role="user", content=request.message))
    
    # 處理訊息
    result = conversation_graph.process_message(context, request.message)
    updated_context = result["context"]
    
    # 生成回應
    if updated_context.state == ConversationState.CLARIFYING:
        response_message = result.get("clarification_question", "能否請您更具體地說明？")
        suggestions = result.get("suggestions", [])
    else:
        response_message = result.get("final_answer", "處理中...")
        suggestions = []
    
    return ConversationResponse(
        session_id=session_id,
        message=response_message,
        state=updated_context.state,
        suggestions=suggestions,
        is_complete=(updated_context.state == ConversationState.COMPLETED)
    )

@app.get("/api/sessions/{session_id}")
async def get_session(session_id: str):
    """獲取對話歷史"""
    if session_id not in active_conversations:
        raise HTTPException(status_code=404, detail="Session not found")
    
    context = active_conversations[session_id]
    return {
        "session_id": session_id,
        "messages": context.messages,
        "state": context.state.value,
        "clarification_count": context.clarification_count
    }

if __name__ == "__main__":
    uvicorn.run(app, host="0.0.0.0", port=8000)

        ws.onopen = () => {
            statusDiv.textContent = '已連接';
            sendBtn.disabled = false;
        };
        
        ws.onmessage = (event) => {
            const data = JSON.parse(event.data);
            hideTyping();
            
            // 添加助手訊息
            addMessage(data.message, 'assistant');
            
            // 更新狀態
            if (data.state) {
                updateStatus(data.state);
            }
            
            // 顯示建議
            if (data.suggestions && data.suggestions.length > 0) {
                showSuggestions(data.suggestions);
            } else {
                suggestionsDiv.innerHTML = '';
            }
            
            // 顯示信心分數
            if (data.confidence_score !== undefined) {
                const confidence = Math.round(data.confidence_score * 100);
                const confidenceDiv = document.createElement('div');
                confidenceDiv.className = 'confidence-indicator';
                confidenceDiv.textContent = `信心度: ${confidence}%`;
                messagesDiv.lastElementChild.appendChild(confidenceDiv);
            }
        };
        
        ws.onerror = (error) => {
            console.error('WebSocket 錯誤:', error);
            statusDiv.textContent = '連接錯誤';
        };

            statusDiv.textContent = '已斷開連接';
